function is_guest(l) local s = l:lower() if s:find("192.168.5.", 1, true) or s:find("br-guest", 1, true) then return true end for v in pairs(guest_ifaces) do if s:find(v, 1, true) then return true end end return false end local function is_lan(l) local s = l:lower() return s:find("192.168.0.", 1, true) or s:find("br-lan", 1, true) or s:find("ra0", 1, true) or s:find("rai0", 1, true) or s:find("eth0", 1, true) or s:find("rndis0", 1, true) end local function in_scope(l) if scope == "guest" then return is_guest(l) end if scope == "lan" then return is_lan(l) end return true end local raw = sys.exec("logread 2>/dev/null | tail -n 4000") or "" local entries, counts = {}, {} local total = 0 local function classify(l) local s = l:lower() if s:find("dropbear", 1, true) or s:find("br-", 1, true) then return "Interfaces" end return nil end if s:find("wgvpn", 1, true) or s:find("wghome", 1, true) or s:find("wireguard", 1, true) or s:find("wg-quick", 1, true) then return "VPN" end if s:find("ptrl", 1, true) or s:find("firewall", 1, true) or s:find("reject", 1, true) or s:find("dropped", 1, true) or s:find("drop packet", 1, true) then return "Firewall" end if s:find("hostapd", 1, true) or s:find("wpa_supplicant", 1, true) or s:find("wapp:", 1, true) or s:find("wapp[", 1, true) or s:find("wlan", 1, true) or s:find("wireless", 1, true) or s:find("ra0", 1, true) or s:find("rai0", 1, true) or s:find("apcli", 1, true) then return "WiFi" end for v in pairs(guest_ifaces) do if s:find(v, 1, true) then return "WiFi" end end if s:find("dnsmasq", 1, true) or s:find("odhcpd", 1, true) or s:find("dhcp", 1, true) then return "DHCP/DNS" end if s:find("netifd", 1, true) then return "Interfaces" end return nil end for line in raw:gmatch("[^\n]+") do local cat = classify(line) if cat and in_scope(line) then local ts, msg = line:match("^(%a+%s+%a+%s+%d+%s+%d+:%d+:%d+)%s+%d+%s+(.*)$") if not ts then ts, msg = line:match("^(%a+%s+%d+%s+%d+:%d+:%d+)%s+(.*)$") end if not ts then ts, msg = "", line end entries[#entries + 1] = { ts = ts, msg = msg, cat = cat } counts[cat] = (counts[cat] or 0) + 1 total = total + 1 end end local cat_list = { "Interfaces", "DHCP/DNS", "WiFi", "Firewall", "VPN" } %>
