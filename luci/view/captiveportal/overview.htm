<%+header%> <% local uci = require "luci.model.uci".cursor() local fs = require "nixio.fs" local sys = require "luci.sys" local util = require "luci.util" local pcdata = util.pcdata local function get(opt, default) local v = uci:get("captiveportal", "main", opt) if v == nil or v == "" then return default end return v end local function count_set(name) local raw = sys.exec("ipset list " .. name .. " 2>/dev/null | awk '/^Number of entries/ {print $4}'") or "0" return tonumber((raw:gsub("%s+", ""))) or 0 end local function read_dir(path) local out = {} local ok, d = pcall(fs.dir, path) if not ok or not d then return out end for name in d do if name ~= "." and name ~= ".." then out[#out + 1] = name end end return out end local function bridge_members(dev) local out = read_dir("/sys/class/net/" .. dev .. "/brif") table.sort(out) return out end local function is_bridge(dev) return fs.stat("/sys/class/net/" .. dev .. "/bridge") ~= nil end local enabled = get("enabled", "1") local port = get("port", "8088") local title = get("title", "Guest Wi-Fi") local subtitle = get("subtitle", "Free guest Wi-Fi") local intro = get("intro", "") local minutes = get("session_minutes", "240") local exempt = get("exempt", "") local gate_iface = get("gate_iface", "br-lan") if not gate_iface:match("^[%w_][%w_%.%-:]*$") or gate_iface:find("%.%.") then gate_iface = "br-lan" end local terms_fl = get("terms_file", "/etc/captiveportal.terms.txt") local log_file = get("log_file", "/data/portal/accept.log") local lan = uci:get("network", "lan", "ipaddr") or "192.168.0.1" local saved = luci.http.formvalue("saved") -- MAC vendor table (refreshed by /usr/sbin/oui-update) local oui = {} local oui_f = io.open("/data/oui/oui.tsv", "r") if oui_f then for line in oui_f:lines() do local p, v = line:match("^(%x+)\t(.+)$") if p then oui[p] = v end end oui_f:close() end local function vendor(mac) local s = (mac or ""):lower():gsub(":", "") return oui[s:sub(1,10)] or oui[s:sub(1,9)] or oui[s:sub(1,7)] or oui[s:sub(1,6)] or "-" end local function iface_ip(dev) local out = sys.exec("ip addr show dev " .. dev .. " (" .. why .. ")" }) seen[gate_iface] = true end local function rank(o) if o.dev == "br-lan" then return 0 end if o.dev:match("^br%-") then return 1 end if net_of[o.dev] then return 2 end return 3 end table.sort(options, function(a, b) local ra, rb = rank(a), rank(b) if ra ~= rb then return ra < rb end return a.dev < b.dev end) local function iface_label(dev) local net = net_of[dev] local label = pretty[net or ""] or net or dev if net then label = label .. " (" .. dev .. ")" end local mem = bridge_members(dev) if #mem > 0 then label = label .. " — " .. table.concat(mem, ", ") end return label end local selected_label = iface_label(gate_iface) local selected_member = member_of[gate_iface] local terms = "" if fs.access(terms_fl) then terms = fs.readfile(terms_fl) or "" end -- last acceptances / requests -- accept.log: ts | ip | mac | country | target | ua | vendor -- requests: ts | ip | mac | country | mode | ua local function split_line(line) local f = {} for v in (line .. "|"):gmatch("([^|]*)|") do f[#f + 1] = (v:gsub("^%s+", ""):gsub("%s+$", "")) end if #f >= 6 then return { ts = f[1], ip = f[2], mac = f[3], cc = f[4], target = f[5], ua = f[6], vendor = (f[7] and f[7] ~= "") and f[7] or vendor(f[3]) } elseif #f >= 5 then return { ts = f[1], ip = f[2], mac = f[3], cc = "-", target = f[4], ua = f[5], vendor = vendor(f[3]) } end end local function tail_lines(path, n) local out = {} if not fs.access(path) then return out end local f = io.open(path, "r") if not f then return out end local lines = {} for line in f:lines() do if line ~= "" then lines[#lines + 1] = line end end f:close() for i = math.max(1, #lines - n + 1), #lines do local t = split_line(lines[i]) if t then out[#out + 1] = t end end return out end local accepts = tail_lines(log_file, 25) local requests = tail_lines("/tmp/portal-access.log", 8) local today, count = os.date("%Y-%m-%d"), 0 if fs.access(log_file) then local f = io.open(log_file, "r") if f then for line in f:lines() do if line:sub(1, 10) == today then count = count + 1 end end f:close() end end %>
